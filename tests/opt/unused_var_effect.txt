// read and a printing call keep their declarations; quiet prints nothing,
// so its results are dropped; the global last is only written, so it goes
// together with the assignment inside remember
:||::::: loud (::|::::: v) ::|:::::
{
    :||:::|: v
    |||||||| v
}
:||::::: quiet (::|::::: v) ::|:::::
{
    |||||||| (v * 2)
}
:||::::: remember (::|::::: v) ::::::::
{
    last = v
}
