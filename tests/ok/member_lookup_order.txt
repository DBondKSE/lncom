// in a member function: parameters and locals first, then fields, then globals
:::::||: P
{
    ::|::::: x

    :||::::: param (::|::::: x) ::|:::::
    {
        |||||||| x
    }

    :||::::: local () ::|:::::
    {
        ::|::::: x {4}
        |||||||| x
    }

    :||::::: field () ::|:::::
    {
        |||||||| x
    }

    :||::::: global () ::|:::::
    {
        |||||||| y
    }
}
::|::::: x {100}
::|::::: y {200}
:::::||: P p {3}
:||:::|: p.param(1)
:||:::|: p.local()
:||:::|: p.field()
|||||||| p.global()
