// a function may call a function declared below it
:||::::: is_even (::|::::: n) ::::::|:
{
    ||||:::: n == 0
    {
        |||||||| :::::::|
    }
    |||||||| is_odd(n - 1)
}
:||::::: is_odd (::|::::: n) ::::::|:
{
    ||||:::: n == 0
    {
        |||||||| ::::::::
    }
    |||||||| is_even(n - 1)
}
:||:::|: is_even(10)
|||||||| is_odd(7)
