// an array is copied when passed and when returned: total does not see
// later changes, and bumped returns a new array
:||::::: total (::::|:|:@::|:::::@3 xs) ::|:::::
{
    |||||||| (xs@0 + xs@1 + xs@2)
}
:||::::: bumped (::::|:|:@::|:::::@3 xs) ::::|:|:@::|:::::@3
{
    |:::|:|: r {xs}
    r@0 = r@0 + 100
    |||||||| r
}
|:::|:|: a {::|:::::@3@2}
::::|:|: b {bumped(a)}
a@1 = 5
:||:::|: a
:||:::|: b
:||:::|: total(a)
:||:::|: bumped(b)@0
|:::|:|: c {a}
