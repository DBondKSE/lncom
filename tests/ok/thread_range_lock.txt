// the user types the thread count; thread k adds k*1000+1..(k+1)*1000
// to the shared total inside the lock
::::||:: m
:||::::: part (::|::::: k, ::|::::: size) ::::::::
{
    ||:::::: s {0}
    |:|::::: i {k * size + 1}
    |:|:|:|: i != (k + 1) * size + 1
    {
        s = s + i
        i = i + 1
    }
    ::||||:: m
    {
        total = total + s
    }
}
||:::::: total {0}
::|::::: count {:||::::|}
:||:::||
{
    @count part(@, 1000)
}
